GBuffer列表：
GBuffer0 - WorldPosition + Depth // RGBA32F (需要高精度存储位置)
.r = worldPos.x    // 世界坐标X
.g = worldPos.y    // 世界坐标Y  
.b = worldPos.z    // 世界坐标Z
.a = linearDepth   // 线性深度 [0.0-1.0]

GBuffer1 - BaseColor + MaterialType // RGBA16F
.r = baseColor.r
.g = baseColor.g  
.b = baseColor.b
.a = materialType  // 材质类型标识符

GBuffer2 - MetallicRoughness + AO // RGBA16F
.r = metallic      // 金属度 [0.0-1.0]
.g = roughness     // 粗糙度 [0.0-1.0] 
.b = occlusion     // 环境光遮蔽 [0.0-1.0]
.a = reserved      // 保留位

GBuffer3 - Normal + NormalScale // RGBA16F  
.r = normal.x      // 世界空间法线X [-1.0-1.0]
.g = normal.y      // 世界空间法线Y [-1.0-1.0]
.b = normal.z      // 世界空间法线Z [-1.0-1.0]
.a = normalScale   // 法线贴图强度 [0.0-1.0]

GBuffer4 - Emission + Alpha // RGBA16F
.r = emission.r    // 自发光颜色R
.g = emission.g    // 自发光颜色G
.b = emission.b    // 自发光颜色B
.a = alpha         // 透明度 [0.0-1.0]（用于后续前向渲染判断）

GBuffer5 - NPRParameters（PBR模式下写入默认值） // RGBA16F
.r = rampThreshold    // 主要光照分界点--色阶阈值 [0.0-1.0]
.g = rampSmoothness   // 光照过渡平滑度--色阶平滑度 [0.0-0.2]
.b = specularSize     // 高光大小/强度--高光尺寸 [0.0-1.0]
.a = outlineWidth     // 描边粗细--描边宽度 [0.0-0.1]

GBuffer6 - NPRColors（PBR模式下写入默认值） // RGBA16F  
.r = shadowTint.r     // 阴影色调R--阴影区域颜色
.g = shadowTint.g     // 阴影色调G  
.b = shadowTint.b     // 阴影色调B
.a = rimPower         // 边缘光衰减--边缘光强度 [0.0-5.0]
